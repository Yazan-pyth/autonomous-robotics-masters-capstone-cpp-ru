cmake_minimum_required(VERSION 3.16)
project(capstone_bringup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)

add_executable(clock_probe src/clock_probe.cpp)
target_include_directories(clock_probe PRIVATE include)
ament_target_dependencies(clock_probe rclcpp rosgraph_msgs)
target_compile_options(clock_probe PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS clock_probe DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(clock_progress_test test/clock_progress_test.cpp)
  target_include_directories(clock_progress_test PRIVATE include)
  target_compile_options(clock_progress_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME clock_progress COMMAND clock_progress_test)
endif()
ament_package()
